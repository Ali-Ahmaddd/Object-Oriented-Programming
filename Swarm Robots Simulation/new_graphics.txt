    void startGraphics()
    {
        const int SIZE = 15;
        const int CELL = 48;
        const int gridPixel = SIZE * CELL;
        const int winW = 1440;
        const int winH = 900;
        const int MARGIN_LEFT = 370;
        const int MARGIN_TOP = 100;

        sf::RenderWindow window(sf::VideoMode(sf::Vector2u((unsigned)winW, (unsigned)winH)), "Swarm Robotics Control Dashboard");
        
        sf::Font font;
        if(!font.openFromFile("C:/Windows/Fonts/arialbd.ttf") && !font.openFromFile("C:/Windows/Fonts/arial.ttf"))
        {
            logger.error("Failed to load font");
        }

        sf::Clock clock;
        float updateInterval = 0.5f;
        float accumulator = 0.0f;
        float animTime = 0.0f;

        // Custom UI drawing helper
        auto drawPanel = [&](sf::RenderWindow& w, float x, float y, float width, float height, string title) {
            sf::RectangleShape rect(sf::Vector2f(width, height));
            rect.setPosition(sf::Vector2f(x, y));
            rect.setFillColor(sf::Color(20, 28, 38));
            rect.setOutlineThickness(1.5f);
            rect.setOutlineColor(sf::Color(40, 50, 65));
            w.draw(rect);
            
            if (title != "") {
                sf::Text t(font, title, 14);
                t.setFillColor(sf::Color(80, 160, 220));
                t.setPosition(sf::Vector2f(x + 15, y + 15));
                w.draw(t);
            }
        };

        auto drawRobotFallback = [&](sf::RenderWindow& w, const sf::Vector2f& pos, bool collector, float size) {
            sf::Color mainC = collector ? sf::Color(60, 140, 240) : sf::Color(80, 200, 80);
            sf::CircleShape halo(size * 0.6f);
            int haloA = 50 + int(30.0f * (std::sin(animTime * 4.0f) + 1.0f) / 2.0f);
            halo.setFillColor(sf::Color(mainC.r, mainC.g, mainC.b, haloA));
            halo.setOrigin(sf::Vector2f(halo.getRadius(), halo.getRadius()));
            halo.setPosition(pos);
            w.draw(halo);

            sf::ConvexShape shape;
            shape.setPointCount(3);
            shape.setPoint(0, sf::Vector2f(0, -size*0.4f));
            shape.setPoint(1, sf::Vector2f(size*0.35f, size*0.35f));
            shape.setPoint(2, sf::Vector2f(-size*0.35f, size*0.35f));
            shape.setFillColor(mainC);
            shape.setOrigin(sf::Vector2f(0, 0));
            shape.setPosition(pos);
            w.draw(shape);
        };

        renderPositions.clear();
        trails.clear();
        renderPositions.resize(robots.size());
        trails.resize(robots.size());
        for(size_t i = 0; i < robots.size(); i++)
        {
            float rx = MARGIN_LEFT + robots[i]->getY() * CELL + CELL / 2.0f;
            float ry = MARGIN_TOP + robots[i]->getX() * CELL + CELL / 2.0f;
            renderPositions[i] = sf::Vector2f(rx, ry);
            trails[i].push_back(sf::Vector2f((float)robots[i]->getX(), (float)robots[i]->getY()));
        }

        while(window.isOpen())
        {
            while(true)
            {
                auto ev = window.pollEvent();
                if(!ev) break;
            }

            if(sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Escape)) window.close();

            float dt = clock.restart().asSeconds();
            accumulator += dt;
            animTime += dt;

            if(running && !paused && accumulator >= updateInterval && currentStep < maxSteps)
            {
                update();
                accumulator = 0.0f;
            }

            window.clear(sf::Color(10, 15, 20)); // Deep dark blue background

            // --- LEFT COLUMN ---
            float leftX = 20;
            drawPanel(window, leftX, 20, 320, 300, "GLOBAL STATISTICS");
            drawPanel(window, leftX, 340, 320, 400, "ROBOT TELEMETRY");
            drawPanel(window, leftX, 760, 320, 120, "MISSION OBJECTIVES");

            // Global Stats text
            sf::Text statText(font, "Simulation Time      00:02:46\nCurrent Step           " + to_string(currentStep) + "\nRobots Active          6 / 6\nResources Found     12\nTotal Distance         " + to_string(totalDistance) + " m\nEnergy Consumed   68%", 14);
            statText.setFillColor(sf::Color(200, 210, 220));
            statText.setPosition(sf::Vector2f(leftX + 20, 60));
            statText.setLineSpacing(1.8f);
            window.draw(statText);

            // --- CENTER COLUMN ---
            float centerX = 360;
            // Header
            sf::Text title(font, "SWARM ROBOTICS SIMULATION", 20);
            title.setFillColor(sf::Color(80, 180, 240));
            title.setPosition(sf::Vector2f(centerX + 220, 20));
            window.draw(title);

            // GRID
            sf::RectangleShape gridBgRect(sf::Vector2f(gridPixel, gridPixel));
            gridBgRect.setPosition(sf::Vector2f(MARGIN_LEFT, MARGIN_TOP));
            gridBgRect.setFillColor(sf::Color(20, 30, 40));
            window.draw(gridBgRect);

            // Grid lines
            for(int i = 0; i <= SIZE; i++) {
                sf::RectangleShape hLine(sf::Vector2f(gridPixel, 1));
                hLine.setPosition(sf::Vector2f(MARGIN_LEFT, MARGIN_TOP + i * CELL));
                hLine.setFillColor(sf::Color(40, 60, 80));
                window.draw(hLine);

                sf::RectangleShape vLine(sf::Vector2f(1, gridPixel));
                vLine.setPosition(sf::Vector2f(MARGIN_LEFT + i * CELL, MARGIN_TOP));
                vLine.setFillColor(sf::Color(40, 60, 80));
                window.draw(vLine);
                
                if (i < SIZE) {
                    sf::Text lbl(font, to_string(i), 12);
                    lbl.setFillColor(sf::Color(100, 120, 140));
                    lbl.setPosition(sf::Vector2f(MARGIN_LEFT + i*CELL + 20, MARGIN_TOP - 25));
                    window.draw(lbl);
                    
                    sf::Text lbl2(font, to_string(i), 12);
                    lbl2.setFillColor(sf::Color(100, 120, 140));
                    lbl2.setPosition(sf::Vector2f(MARGIN_LEFT - 25, MARGIN_TOP + i*CELL + 15));
                    window.draw(lbl2);
                }
            }

            // Draw Base
            sf::RectangleShape bShape(sf::Vector2f(CELL*0.6f, CELL*0.6f));
            bShape.setFillColor(sf::Color(150, 50, 200));
            bShape.setPosition(sf::Vector2f(MARGIN_LEFT + baseLocation.second * CELL + CELL*0.2f, MARGIN_TOP + baseLocation.first * CELL + CELL*0.2f));
            window.draw(bShape);

            // Draw Resources
            for(auto& r : resources) {
                if(!r.isCollected()) {
                    sf::CircleShape star(CELL*0.25f);
                    star.setFillColor(sf::Color(255, 215, 0));
                    star.setPosition(sf::Vector2f(MARGIN_LEFT + r.getY() * CELL + CELL*0.25f, MARGIN_TOP + r.getX() * CELL + CELL*0.25f));
                    window.draw(star);
                }
            }

            // Draw Obstacles
            for(auto& o : obstacles) {
                sf::RectangleShape obs(sf::Vector2f(CELL*0.7f, CELL*0.7f));
                obs.setFillColor(sf::Color(100, 110, 120));
                obs.setPosition(sf::Vector2f(MARGIN_LEFT + o.getY() * CELL + CELL*0.15f, MARGIN_TOP + o.getX() * CELL + CELL*0.15f));
                window.draw(obs);
            }

            // Draw Robots and Trails
            for(size_t i = 0; i < robots.size(); i++) {
                sf::Color trailC = robots[i]->getType() == "Explorer" ? sf::Color(80, 200, 80) : sf::Color(60, 140, 240);
                
                // Trail
                for(size_t t = 1; t < trails[i].size(); t++) {
                    float x1 = MARGIN_LEFT + trails[i][t-1].y * CELL + CELL/2.0f;
                    float y1 = MARGIN_TOP + trails[i][t-1].x * CELL + CELL/2.0f;
                    float x2 = MARGIN_LEFT + trails[i][t].y * CELL + CELL/2.0f;
                    float y2 = MARGIN_TOP + trails[i][t].x * CELL + CELL/2.0f;
                    
                    sf::Vertex line[] = {
                        sf::Vertex(sf::Vector2f(x1, y1), trailC),
                        sf::Vertex(sf::Vector2f(x2, y2), trailC)
                    };
                    window.draw(line, 2, sf::PrimitiveType::Lines);
                }

                float targetX = MARGIN_LEFT + robots[i]->getY() * CELL + CELL / 2.0f;
                float targetY = MARGIN_TOP + robots[i]->getX() * CELL + CELL / 2.0f;
                
                renderPositions[i].x += (targetX - renderPositions[i].x) * 0.1f;
                renderPositions[i].y += (targetY - renderPositions[i].y) * 0.1f;
                
                drawRobotFallback(window, renderPositions[i], robots[i]->getType() != "Explorer", CELL*0.8f);
            }

            // Bottom Progress & Legend
            drawPanel(window, centerX, 840, 420, 80, "MISSION PROGRESS");
            drawPanel(window, centerX + 440, 840, 320, 80, "LEGEND");

            // --- RIGHT COLUMN ---
            float rightX = 1120;
            drawPanel(window, rightX, 20, 300, 400, "SYSTEM LOGS");
            drawPanel(window, rightX, 440, 300, 260, "ENERGY LEVELS");
            drawPanel(window, rightX, 720, 300, 160, "CONTROLS");

            // Controls
            sf::RectangleShape btn(sf::Vector2f(80, 30));
            btn.setFillColor(sf::Color(60, 160, 60));
            btn.setPosition(sf::Vector2f(rightX + 20, 780));
            window.draw(btn);
            sf::Text tBtn(font, "Start", 14);
            tBtn.setFillColor(sf::Color::White);
            tBtn.setPosition(sf::Vector2f(rightX + 40, 785));
            window.draw(tBtn);

            window.display();
        }
    }
